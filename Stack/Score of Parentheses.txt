class Solution {
    public int scoreOfParentheses(String s) {
        Stack<Integer> stack=new Stack<>();
        stack.add(0);
        for(char c:s.toCharArray()){
            if(c=='('){
                stack.add(0);
            }else{
                int current=stack.pop();
                if(current==0){
                    current=1;
                }else{
                    current*=2;
                }
                stack.add(stack.pop()+current);

            }
        }
        return stack.pop();
    }
}
