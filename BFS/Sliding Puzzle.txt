773 SLIDING PUZZLE

class Solution {
    public int slidingPuzzle(int[][] board) {
        Queue<int[][]> q=new LinkedList<>();
		Queue<int[][]> vis=new LinkedList<>();
		Queue<Integer> moves=new LinkedList<>();
		int[][] target={{1,2,3},{4,5,0}};
        if(Arrays.deepEquals(board,target)){
            return 0;
        }
		int x=0;
		int y=0;
		for(int i=0;i<2;i++){
		    for(int j=0;j<3;j++){
		        if(board[i][j]==0){
		            x=i;
		            y=j;
		        }
		    }
		}
		q.offer(board);
		moves.offer(0);
		if(x-1>=0){
		    int[][] b=new int[2][3];
		    for(int i=0;i<2;i++){
		        b[i]=Arrays.copyOf(board[i],3);
		    }
		    int temp=b[x-1][y];
		    b[x-1][y]=b[x][y];
		    b[x][y]=temp;
		    q.offer(b);
		    moves.offer(1);
		}
		if(y-1>=0){
		    int[][] b=new int[2][3];
		    for(int i=0;i<2;i++){
		        b[i]=Arrays.copyOf(board[i],3);
		    }
		    int temp=b[x][y-1];
		    b[x][y-1]=b[x][y];
		    b[x][y]=temp;
		    q.offer(b);
		    moves.offer(1);
		}
		if(x+1<2){
		    int[][] b=new int[2][3];
		    for(int i=0;i<2;i++){
		        b[i]=Arrays.copyOf(board[i],3);
		    }
		    int temp=b[x+1][y];
		    b[x+1][y]=b[x][y];
		    b[x][y]=temp;
		    q.offer(b);
		    moves.offer(1);
		}
		if(y+1<3){
		    int[][] b=new int[2][3];
		    for(int i=0;i<2;i++){
		        b[i]=Arrays.copyOf(board[i],3);
		    }
		    int temp=b[x][y+1];
		    b[x][y+1]=b[x][y];
		    b[x][y]=temp;
		    q.offer(b);
		    moves.offer(1);
		}
		vis.add(board);
		while(!q.isEmpty()){
		    int[][] state=q.poll();
		    int step=moves.poll();
		    boolean f=false;
		    for(int[][] row:vis){
		        if(Arrays.deepEquals(state,row)){
		            f=true;
		            break;
		        }
		    }
		    if(f)
		        continue;
		        
		    if(Arrays.deepEquals(state,target)){
		        return step;
		    }
		    vis.offer(state);
		    for(int i=0;i<2;i++){
		        for(int j=0;j<3;j++){
		            if(state[i][j]==0){
		                x=i;
		                y=j;
		            }
		        }
		    }
		    if(x-1>=0){
		        int[][] b=new int[2][3];
		        for(int i=0;i<2;i++){
		            b[i]=Arrays.copyOf(state[i],3);
		        }
		        int temp=b[x-1][y];
		        b[x-1][y]=b[x][y];
		        b[x][y]=temp;
		        q.offer(b);
		        moves.offer(step+1);
		    }
		if(y-1>=0){
		    int[][] b=new int[2][3];
		    for(int i=0;i<2;i++){
		        b[i]=Arrays.copyOf(state[i],3);
		    }
		    int temp=b[x][y-1];
		    b[x][y-1]=b[x][y];
		    b[x][y]=temp;
		    q.offer(b);
		    moves.offer(step+1);
		}
		if(x+1<2){
		    int[][] b=new int[2][3];
		    for(int i=0;i<2;i++){
		        b[i]=Arrays.copyOf(state[i],3);
		    }
		    int temp=b[x+1][y];
		    b[x+1][y]=b[x][y];
		    b[x][y]=temp;
		    q.offer(b);
		    moves.offer(step+1);
		}
		if(y+1<3){
		    int[][] b=new int[2][3];
		    for(int i=0;i<2;i++){
		        b[i]=Arrays.copyOf(state[i],3);
		    }
		    int temp=b[x][y+1];
		    b[x][y+1]=b[x][y];
		    b[x][y]=temp;
		    q.offer(b);
		    moves.offer(step+1);
		}
		}
        return -1;
    }
}
