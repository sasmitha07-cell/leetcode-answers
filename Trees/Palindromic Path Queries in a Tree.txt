3841. Palindromic Path Queries in a Tree


import java.util.*;

class Solution {
    public List<Boolean> palindromePath(
            int n, int[][] edges, String s, String[] queries) {

        List<Integer>[] adj = new ArrayList[n];

        for (int i = 0; i < n; i++) {
            adj[i] = new ArrayList<>();
        }

        for (int[] e : edges) {
            adj[e[0]].add(e[1]);
            adj[e[1]].add(e[0]);
        }

        int LOG = 1;
        while ((1 << LOG) <= n) LOG++;

        int[][] up = new int[LOG][n];
        int[] tin = new int[n];
        int[] tout = new int[n];
        int[] parent = new int[n];
        int[] depth = new int[n];
        int[] order = new int[n];
        int[] stack = new int[n];
        int[] next = new int[n];

        int top = 0, timer = 0;
        stack[top++] = 0;
        parent[0] = 0;
        tin[0] = timer;
        order[timer++] = 0;

        while (top > 0) {
            int u = stack[top - 1];

            if (next[u] < adj[u].size()) {
                int v = adj[u].get(next[u]++);

                if (v == parent[u]) continue;

                parent[v] = u;
                depth[v] = depth[u] + 1;
                tin[v] = timer;
                order[timer++] = v;
                stack[top++] = v;
            } else {
                tout[u] = timer;
                top--;
            }
        }

        for (int i = 0; i < n; i++) {
            up[0][i] = parent[i];
        }

        for (int j = 1; j < LOG; j++) {
            for (int i = 0; i < n; i++) {
                up[j][i] = up[j - 1][up[j - 1][i]];
            }
        }

        int[] value = new int[n];
        int[] prefix = new int[n];

        for (int i = 0; i < n; i++) {
            value[i] = 1 << (s.charAt(i) - 'a');
        }

        for (int u : order) {
            prefix[u] = value[u];

            if (u != 0) {
                prefix[u] ^= prefix[parent[u]];
            }
        }

        Fenwick bit = new Fenwick(n);
        List<Boolean> ans = new ArrayList<>();

        for (String q : queries) {
            String[] parts = q.split(" ");
            int u = Integer.parseInt(parts[1]);

            if (parts[0].equals("update")) {
                int newValue = 1 << (parts[2].charAt(0) - 'a');
                int delta = value[u] ^ newValue;

                bit.update(tin[u], delta);
                bit.update(tout[u], delta);

                value[u] = newValue;
            } else {
                int v = Integer.parseInt(parts[2]);
                int lca = getLCA(u, v, depth, up);

                int mask = (prefix[u] ^ bit.query(tin[u]))
                         ^ (prefix[v] ^ bit.query(tin[v]))
                         ^ value[lca];

                ans.add((mask & (mask - 1)) == 0);
            }
        }

        return ans;
    }

    private int getLCA(int u, int v, int[] depth, int[][] up) {
        if (depth[u] < depth[v]) {
            int temp = u;
            u = v;
            v = temp;
        }

        int diff = depth[u] - depth[v];

        for (int j = 0; j < up.length; j++) {
            if ((diff & (1 << j)) != 0) {
                u = up[j][u];
            }
        }

        if (u == v) return u;

        for (int j = up.length - 1; j >= 0; j--) {
            if (up[j][u] != up[j][v]) {
                u = up[j][u];
                v = up[j][v];
            }
        }

        return up[0][u];
    }

    static class Fenwick {
        int[] tree;
        int n;

        Fenwick(int n) {
            this.n = n;
            tree = new int[n + 2];
        }

        void update(int index, int value) {
            for (int i = index + 1; i <= n + 1; i += i & -i) {
                tree[i] ^= value;
            }
        }

        int query(int index) {
            int result = 0;

            for (int i = index + 1; i > 0; i -= i & -i) {
                result ^= tree[i];
            }

            return result;
        }
    }
}
